```swift
import XCTest
@testable import GraphiteCore

/// Reproductions of claimed bugs from a code-reading bug hunt. Each test asserts the claimed
/// WRONG output, so a passing test means the bug is still present. Claims 37, 61, 155 and 246
/// live in GraphiteUI, MermaidDiagrams or SwiftUI and cannot be run from this target.
final class BugHuntReproductionTests: XCTestCase {
    private func report(_ claim: Int, _ actual: Any) { print("BUGHUNT_ACTUAL \(claim): \(actual)") }

    private func applied(_ edit: MarkdownTextEdit, to text: String) -> String {
        (text as NSString).replacingCharacters(in: edit.range, with: edit.replacement)
    }

    func test003_boldInsideBold() {
        let text = "**hello world**"
        let edit = MarkdownEditing.togglingWrap("**", closingMarker: "**", in: text as NSString, selection: NSRange(location: 10, length: 0))
        let result = applied(edit, to: text)
        report(3, result)
        XCTAssertEqual(result, "**hello **world****")
    }

    func test035_returnInMathBlockAddsListMarker() {
        let text = "$$\n+ b\n$$"
        let edit = MarkdownEditing.continuingList(in: text as NSString, selection: NSRange(location: 6, length: 0), indentUnit: "    ")
        report(35, edit.map { "\($0)" } ?? "nil")
        XCTAssertNotNil(edit)
        XCTAssertTrue(edit?.replacement.contains("+ ") ?? false)
    }

    func test036_headingOnQuotedLine() {
        let text = "> text"
        let edit = MarkdownEditing.settingHeading(level: 2, in: text as NSString, selection: NSRange(location: 3, length: 0))
        let result = applied(edit, to: text)
        report(36, result)
        XCTAssertEqual(result, "## > text")
    }

    func test055_secondFootnoteParagraphLost() {
        let text = "Ref[^1]\n\n[^1]: First paragraph.\n\n    Second paragraph, indented four spaces.\n"
        let parsed = Footnotes.parse(text)
        report(55, parsed.definitions.map { $0.text })
        XCTAssertEqual(parsed.definitions.count, 1)
        XCTAssertFalse(parsed.definitions[0].text.contains("Second paragraph"))
    }

    func test056_headingAnchorKeepsColon() {
        let heading = NotePreviewDocument.anchor(forHeading: "Step 1: Setup")
        let link = NotePreviewDocument.anchor(forHeading: "Step 1 Setup")
        report(56, "\(heading) vs \(link)")
        XCTAssertNotEqual(heading, link)
    }

    func test094_renameToLinkBreakingNameAccepted() {
        let problem = FileNameRules.problem(with: "photo #2.png", isNote: false)
        let rewritten = LinkRewriter.replacingPath(inLinkText: "![[photo.png]]", isWiki: true, newPathPart: "photo #2.png")
        report(94, "problem=\(String(describing: problem)) rewritten=\(String(describing: rewritten))")
        XCTAssertNil(problem)
        XCTAssertEqual(rewritten, "![[photo #2.png]]")
    }

    func test127_typingAfterUnterminatedFrontmatterDestroysIt() {
        let before = "---\ntitle: x\n---"
        let after = before + "H"
        let lengthBefore = FrontmatterLocator.length(in: before as NSString)
        let lengthAfter = FrontmatterLocator.length(in: after as NSString)
        report(127, "before=\(lengthBefore) of \((before as NSString).length), after=\(lengthAfter)")
        XCTAssertEqual(lengthBefore, (before as NSString).length)
        XCTAssertEqual(lengthAfter, 0)
    }

    func test130_headingStretchesSelection() {
        let text = "abc def"
        let edit = MarkdownEditing.settingHeading(level: 2, in: text as NSString, selection: NSRange(location: 0, length: 3))
        report(130, "\(edit.selectionAfter) replacement=\(edit.replacement)")
        XCTAssertNotEqual(edit.selectionAfter.length, 3)
        XCTAssertGreaterThan(edit.selectionAfter.length, 3)
    }

    func test131_escapedStarsDrawnItalic() {
        let source = "\\*not italic\\*" as NSString
        let spans = MarkdownStyleScanner.spans(in: source, range: NSRange(location: 0, length: source.length))
        report(131, spans.map { "\(source.substring(with: $0.range)):\($0.style)" })
        XCTAssertTrue(spans.contains { $0.style == .emphasis })
    }

    func test144_inlineFootnoteWithLinkCutAtFirstBracket() {
        let parsed = Footnotes.parse("Claim^[see [[Source note]] for details] more text")
        report(144, parsed.references.map { $0.inlineText ?? "nil" })
        XCTAssertEqual(parsed.references.first?.inlineText, "see [[Source note")
    }

    func test146_calloutInListItemNotRecognized() {
        let header = MarkdownQuoteSyntax.calloutHeader(in: "  > [!note] Title")
        let blocks = NotePreviewDocument.blocks(from: "- a\n  > [!note] Title\n  > body")
        let topLevel = NotePreviewDocument.blocks(from: "   > [!tip] Tip\n   > body")
        func hasCallout(_ list: [NotePreviewBlock]) -> Bool { list.contains { if case .callout = $0 { true } else { false } } }
        report(146, "header=\(String(describing: header)) list=\(blocks) indented=\(topLevel)")
        XCTAssertNil(header)
        XCTAssertFalse(hasCallout(blocks))
        XCTAssertFalse(hasCallout(topLevel))
    }

    func test149_calloutEndsAtLazyContinuation() {
        let blocks = NotePreviewDocument.blocks(from: "> [!note] T\n> line one\nline two (no >)\n\nnext paragraph")
        report(149, blocks)
        guard case .callout(_, _, _, let body)? = blocks.first else { return XCTFail("no callout first: \(blocks)") }
        XCTAssertFalse("\(body)".contains("line two"))
        XCTAssertTrue(blocks.dropFirst().contains { "\($0)".contains("line two") })
    }

    func test170_momentFormatsWrittenLiterally() {
        let date = Date(timeIntervalSince1970: 1_793_534_400) // 2026-11-01 12:00 UTC
        let utc = TimeZone(identifier: "UTC")!
        let short = MomentDateFormat.string(from: date, format: "ll", timeZone: utc)
        let escaped = MomentDateFormat.string(from: date, format: "YYYY-MM-DD \\W\\e\\e\\k", timeZone: utc)
        let template = MomentDateFormat.string(from: date, format: "lll", timeZone: utc)
        report(170, "ll=\(short) lll=\(template) escaped=\(escaped)")
        XCTAssertEqual(short, "ll")
    }

    func test191_obsidianNewIgnoresPath() {
        let url = URL(string: "obsidian://new?vault=Physics&path=/full/path/Physics/Notes/Idea&content=hello")!
        let uri = VaultURI(url: url)
        report(191, uri.map { "\($0.action)" } ?? "nil")
        guard case .new(let name, let file, _, _, _)? = uri?.action else { return XCTFail("not a new action") }
        XCTAssertNil(name)
        XCTAssertNil(file)
    }

    func test169_graphFilterIgnoresOperators() {
        let path = "Notes/lecture.md"
        let results = ["-archive", "file:lecture", "lecture OR exam"].map { GraphSearch($0).matches(path: path, tags: []) }
        report(169, results)
        XCTAssertEqual(results, [false, false, false])
    }

    func test034_markdownLinkWithParenthesesOpensWrongPage() {
        let line = "[A](https://en.wikipedia.org/wiki/A_(B))"
        let located = LinkLocator.locatedLink(in: line, at: 5, includesEnd: false)
        let angle = LinkLocator.locatedLink(in: "[x](<https://a.b/c_(d)>)", at: 5, includesEnd: false)
        let titled = LinkLocator.locatedLink(in: "[x](https://a.b \"Title\")", at: 5, includesEnd: false)
        report(34, "paren=\(String(describing: located?.link)) angle=\(String(describing: angle?.link)) titled=\(String(describing: titled?.link))")
        XCTAssertEqual(located?.link, .web(URL(string: "https://en.wikipedia.org/wiki/A_(B")!))
    }

    // Not runnable from GraphiteCoreTests (GraphiteUI, MermaidDiagrams or SwiftUI): 37, 61, 155, 246.
}
```
